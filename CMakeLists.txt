cmake_minimum_required(VERSION 3.25)
project(AudioWire VERSION 0.1.0 LANGUAGES C CXX)
if(MSVC)
    add_compile_options(/utf-8)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
option(AUDIOWIRE_BUILD_APP "Build the native Windows/iOS application" ON)
option(AUDIOWIRE_BUILD_TESTS "Build hardware-independent tests" ON)
set(AUDIOWIRE_VENDOR_DIR "" CACHE PATH "Optional directory of checked-out dependencies")
include(cmake/Dependencies.cmake)
add_library(audiowire_core
    core/Pairing.cpp core/SecureTransport.cpp core/Session.cpp core/Control.cpp
    core/Transport.cpp)
target_include_directories(audiowire_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(audiowire_core PUBLIC sodium aw_aoo)
if(MSVC)
    target_compile_options(audiowire_core PRIVATE /W4)
else()
    target_compile_options(audiowire_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(AUDIOWIRE_BUILD_TESTS)
    enable_testing()
    add_executable(core_tests tests/CoreTests.cpp)
    target_link_libraries(core_tests PRIVATE audiowire_core)
    add_test(NAME core COMMAND core_tests)
endif()
if(AUDIOWIRE_BUILD_APP)
    include(cmake/Application.cmake)
endif()

if(AUDIOWIRE_BUILD_TESTS AND WIN32)
    add_executable(driver_abi_tests tests/DriverAbiTests.cpp)
    target_include_directories(driver_abi_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    add_test(NAME driver_abi COMMAND driver_abi_tests)
endif()
